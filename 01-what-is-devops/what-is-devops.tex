    \begin{frame}{Какую проблему мы обсуждаем?}
        \textbf{Промышленная разработка ПО:}
        \begin{enumerate}
            \item \textbf{Организационно-процессная} составляющая
            \item \textbf{Техническая} составляющая
        \end{enumerate}

        \vspace{0.5cm}

        Между разработкой и эксплуатацией находится важный, \\
        но часто \textbf{``невидимый''} слой — именно он в центре нашего курса
    \end{frame}

    \begin{frame}{Зачем новые подходы?}
        \textbf{Индустрия разработки ПО изменилась:}
        \begin{itemize}[<+->]
        \item Программы стали \textbf{сложнее}
        \item Работают как \textbf{веб-сервисы 24/7}
        \item Бизнес требует \textbf{быстрых обновлений}
        \item Раньше: релиз раз в несколько месяцев
        \item Сейчас: обновления каждую неделю или чаще
        \end{itemize}
    \end{frame}

    \begin{frame}{Что требует такая скорость?}
        \begin{itemize}
            \item \textbf{Высокая автоматизация}
            \item \textbf{Тесное взаимодействие} между командами
            \item \textbf{Быстрое внесение} и проверка изменений
        \end{itemize}

        \vspace{0.5cm}

        \textbf{Традиционные методы} с длительными циклами \\
        и раздельной работой команд \textbf{не успевают}
    \end{frame}

    \begin{frame}{Почему мы здесь?}
        \textbf{Сегодня любой инженер} — разработчик, тестировщик, админ — \\
        не может качественно работать, не понимая:

        \vspace{0.5cm}

        \begin{itemize}
            \item Как код \textbf{тестируется}
            \item Как код \textbf{деплоится}
            \item Как код \textbf{масштабируется}
            \item Как код \textbf{мониторится}
        \end{itemize}

        \vspace{0.5cm}

        \textbf{Цель курса:} понять современный процесс разработки \\
        и практики, делающие его быстрым, надёжным и гибким
    \end{frame}

    \begin{frame}{План курса}
        \begin{enumerate}
            \item Введение в DevOps: философия и культура
            \item Системы контроля версий и Git workflows
            \item CI/CD: концепции и инструменты
            \item Инфраструктура как код (IaC)
            \item Контейнеризация и оркестрация
            \item Мониторинг и логирование
            \item DevSecOps и безопасность в пайплайнах
            \item Deployment стратегии и практический проект
        \end{enumerate}
    \end{frame}

        \end{center}

        \begin{frame}{Кейс 1: Пятничный релиз}
            Пятница. Вечер. В конце рабочего дня разработчик вручную выкатывает билд на прод. Через час пользователи жалуются
            на баги, а восстановить прошлую версию невозможно: она только на ноуте разработчика, который уже уехал.

            \vspace{0.5cm}

            \textbf{Что делать?}
            \begin{enumerate}
                \item Нужно заставить Васю лучше тестировать
                \item Надо было подождать до понедельника
                \item Не хватает процессов и автоматизации
                \item Баги — это нормально, в понедельник починим
            \end{enumerate}
        \end{frame}

        \begin{frame}{Кейс 2: Медленная работа после релиза}
            Сисадмин Петя устанавливает новую версию приложения на прод. Предварительно разработчик Вася и тестировщик Саша тестировали
            эту версию на стейджинге. После релиза пользователи жалуются: приложение начало работать заметно медленнее.
            Оказывается, дело в новом функционале.

            \vspace{0.5cm}

            \textbf{Что делать?}
            \begin{enumerate}
                \item Нужно лучше тестировать, а лучше нанять ещё 2 QA инженеров
                \item Автоматизировать настройку серверов
                \item Купить более мощные сервера для продакшена
                \item Вася должен писать больше тестов
            \end{enumerate}
        \end{frame}

        \begin{frame}{Промежуточный итог}
            \begin{center}
                \Large
                Видите закономерность?

                \vspace{0.5cm}

                В обоих случаях проблема в \textbf{ручных процессах} \\
                и отсутствии \textbf{автоматизации}

                \vspace{0.5cm}

                Но DevOps решает не только технические проблемы...
            \end{center}
        \end{frame}

        \begin{frame}{Кейс 3: Закончилось место на диске}
            Разработчик Вася сделал новую фичу, которая увеличивает использование диска. Сисадмин Петя не знал про новую фичу.
            Через два дня приложение падает — место закончилось.

            \vspace{0.5cm}

            \textbf{Что делать?}
            \begin{enumerate}
                \item Ввести правило: все новые фичи должны согласовываться с Ops за 3 дня
                \item Закупить ``железо`` про запас
                \item Настроить мониторинг и квотирование
                \item Петя должен смотреть все ченжлоги перед релизом
            \end{enumerate}
        \end{frame}

        \begin{frame}{Кейс 5: Важная фича и окно релиза}
            Разработчик Вася за два дня сделал фичу, которая, предположительно, принесет много новых клиентов компании.
            При этом ближайшее окно релиза через 3 недели.

            \vspace{0.5cm}

            \textbf{Что делать?}
            \begin{enumerate}
                \item Разрешить релизить важные фичи вне окна релиза
                \item Мобилизировать тестировщиков с других активностей
                \item Ввести ежедневные планерки для пересмотра приоритетов
                \item Вообще убрать окна релиза и релизить по желанию
            \end{enumerate}
        \end{frame}

                \vspace{0.5cm}

                Но есть еще одна важная проблема

                \vspace{0.5cm}

                \textbf{Что делать, когда что-то идет не так?}
            \end{center}
        \end{frame}

        \begin{frame}{Кейс 6: Удалили таблицу users}
            Неожиданно сайт перестает работать. Выяснилось, что кто-то удалил таблицу users в базе данных.
            Резервная копия устарела, логи не велись.

            \vspace{0.5cm}

            \textbf{Что делать?}
            \begin{enumerate}
                \item Разрешить работать с prod базой только ``роботам``
                \item Поменять все пароли и сообщить их только некоторым синьорам
                \item Вести список ``провинившихся`` разработчиков
                \item Нанять отдельного специалиста DBA
            \end{enumerate}
        \end{frame}

        \begin{frame}{Кейс 7: Вася ушел в отпуск}
            Вася завершил выпуск новой функции приложения, но на следующий день ушёл в планируемый отпуск. Остальные сотрудники не знают, как её тестировать,
            настраивать и использовать, из-за отсутствия документации.

            \vspace{0.5cm}

            \textbf{Что делать?}
            \begin{enumerate}
                \item Внести в общие правила пункт ``Не уходить в отпуск без документации``
                \item Нанять отдельного техписателя для написания документации
                \item Ввести в KPI пункт ``покрытие документацией``
                \item Задача выполнена только тогда, когда она в prod и протестирована
            \end{enumerate}
        \end{frame}
    }

            \Large
            Во всех случаях проблема не в людях

            \vspace{0.5cm}

    \begin{frame}{Три типа проблем}
        \begin{enumerate}
            \item \textbf{Ручные процессы} (кейсы 1, 2, 5)
            \begin{itemize}
                \item Все делается руками вместо автоматизации
                \item Результат: ошибки и нестабильность
            \end{itemize}

            \item \textbf{Изоляция команд} (кейс 3)
            \begin{itemize}
                \item Dev и Ops не понимают друг друга
                \item Результат: проблемы коммуникации
            \end{itemize}

            \item \textbf{Неправильная культура} (кейсы 6, 7)
            \begin{itemize}
                \item Ищут виноватых вместо решения проблем
                \item Результат: повторение ошибок
            \end{itemize}
        \end{enumerate}
    \end{frame}

    \begin{frame}{DevOps - системное решение}
        \begin{center}
            \Large
            \textbf{DevOps решает все три проблемы:}

            \vspace{0.8cm}

            \begin{itemize}
                \item \textbf{Автоматизация} вместо ручной работы
                \item \textbf{Сотрудничество} вместо изоляции команд
                \item \textbf{Культура обучения} вместо поиска виноватых
            \end{itemize}
        \end{center}
    \end{frame}

    \begin{frame}{Кто занимается DevOps?}
        \begin{center}
            \Large
            Вопрос: если DevOps - это культура для всех, \\
            то зачем существует должность \\
            \textbf{``DevOps инженер``}?
        \end{center}
    \end{frame}

    \begin{frame}{DevOps vs DevOps инженер}
        \textbf{DevOps — это культура и практики для всей команды}

        \vspace{0.5cm}

        Но на практике кто-то должен:
        \begin{itemize}
            \item Настраивать CI/CD пайплайны
            \item Управлять инфраструктурой
            \item Внедрять практики автоматизации
            \item Быть связующим звеном между Dev и Ops
        \end{itemize}

        \vspace{0.5cm}

        \textbf{DevOps инженер} — специалист, который помогает командам правильно применять DevOps практики
    \end{frame}

    \begin{frame}{Аналогия с прорабом}
        \begin{center}
            \Large
            Как при строительстве дома:

            \vspace{0.5cm}

            Каждый рабочий знает свое дело — \\
            электрик, сантехник, каменщик

            \vspace{0.5cm}

            Но есть \textbf{прораб}, который координирует всех, \\
            следит за процессом, решает проблемы \\
            между специалистами

            \vspace{0.5cm}

            \textbf{DevOps инженер — прораб в IT-проекте}
        \end{center}
    \end{frame}

    \begin{frame}{Главная мысль}
        \begin{center}
            \Large
            \textbf{Запомните:}

            \vspace{0.8cm}

            DevOps инженер не делает DevOps за всех

            \vspace{0.5cm}

            Он \textbf{координирует процесс} и помогает \\
            команде работать слаженно

            \vspace{0.5cm}

            Прораб не заменяет электрика, \\
            но следит, чтобы работа всех специалистов \\
            не мешала друг другу
        \end{center}
    \end{frame}
