    \begin{frame}{Какую проблему мы обсуждаем?}
        \begin{columns}

            \begin{column}{0.48\textwidth}
                \textbf{Жизненный цикл ПО раньше}
                \begin{itemize}
                    \item Разработка (Development)
                    \item Эксплуатация (Operations)
                \end{itemize}
            \end{column}

            \begin{column}{0.48\textwidth}
                \textbf{Жизненный цикл ПО сейчас}
                \begin{itemize}
                    \item Разработка (Development)
                    \item DevOps
                    \item Эксплуатация (Operations)
                \end{itemize}
            \end{column}

        \end{columns}

        \vspace{0.5cm}

        Между разработкой и эксплуатацией появился новый \enquote{слой} — он в центре нашего курса
    \end{frame}

    \begin{frame}{Откуда появился этот слой ?}
        \textbf{Индустрия разработки ПО изменилась:}
        \begin{itemize}
        \item Программы стали \textbf{сложнее}
        \item Работают как \textbf{веб-сервисы 24/7}
        \item Бизнес требует \textbf{быстрых обновлений}
        \end{itemize}

        \vspace{0.5cm}

        \textbf{Традиционные подходы} не справляются с этими вызовами
    \end{frame}

    \begin{frame}{Ответ на вызовы - DevOps}
        \textbf{DevOps - способ организации работы IT команд}

        Включает:
        \begin{itemize}
            \item Автоматизацию ручных процессов
            \item Сотрудничество разработчиков и админов
            \item Культуру совместной ответственности
        \end{itemize}

        \vspace{0.5cm}

        \textbf{Цель DevOps:} быстрая и надежная доставка ПО
    \end{frame}

    \begin{frame}{Зачем вам это?}
        \begin{enumerate}
            \item Требует работодатель
            \item Делает вас эффективнее
        \end{enumerate}
    \end{frame}

    \begin{frame}{План курса}
        \begin{enumerate}
            \item Введение в DevOps
            \item Системы контроля версий в DevOps
            \item CI/CD: концепции и инструменты
            \item Инфраструктура как код (IaC)
            \item Контейнеризация и оркестрация
            \item Мониторинг и логирование
            \item DevSecOps
            \item Deployment стратегии
        \end{enumerate}
    \end{frame}

    \begin{frame}{Кейс 1: Пятничный релиз}
        Пятница. Вечер. В конце рабочего дня разработчик вручную выкатывает билд на прод. Через час пользователи жалуются
        на баги, а восстановить прошлую версию невозможно: она только на устройстве разработчика, который уже уехал.

        \vspace{0.5cm}

        \textbf{Что делать?}
        \begin{enumerate}
            \item Нужно заставить лучше тестировать
            \item Надо было подождать до понедельника
            \item Не хватает процессов и автоматизации
            \item Баги — это нормально, в понедельник починим
        \end{enumerate}
    \end{frame}

    \begin{frame}{Кейс 2: Медленная работа после релиза}
        Администратор устанавливает новую версию приложения на прод. Предварительно разработчик и тестировщик проверили
        эту версию на стейджинге. После релиза пользователи жалуются: приложение начало работать заметно медленнее.
        Оказывается, дело в новом функционале.

        \vspace{0.5cm}

        \textbf{Что делать?}
        \begin{enumerate}
            \item Нужно лучше тестировать, а лучше нанять ещё 2 тестировщиков
            \item Автоматизировать настройку серверов
            \item Купить более мощные сервера для продакшена
            \item Разработчик должен писать больше тестов
        \end{enumerate}
    \end{frame}

    \begin{frame}{Кейс 3: Закончилось место на диске}
        Разработчик сделал новую фичу, которая увеличивает использование диска. Администратор не знал про новую фичу.
        Через два дня приложение падает — место закончилось.

        \vspace{0.5cm}

        \textbf{Что делать?}
        \begin{enumerate}
            \item Ввести правило: все новые фичи должны согласовываться с Ops за 3 дня
            \item Закупить \enquote{железо} про запас
            \item Настроить мониторинг и квотирование
            \item Администратор должен смотреть все ченжлоги перед релизом
        \end{enumerate}
    \end{frame}

    \begin{frame}{Кейс 4: Важная фича и окно релиза}
        Разработчик за два дня сделал фичу, которая, предположительно, принесет много новых клиентов компании.
        При этом ближайшее окно релиза через 3 недели.

        \vspace{0.5cm}

        \textbf{Что делать?}
        \begin{enumerate}
            \item Разрешить релизить важные фичи вне окна релиза
            \item Мобилизировать тестировщиков с других активностей
            \item Ввести ежедневные планерки для пересмотра приоритетов
            \item Вообще убрать окна релиза и релизить по желанию
        \end{enumerate}
    \end{frame}

    \begin{frame}{Кейс 5: Удалили таблицу users}
        Неожиданно сайт перестает работать. Выяснилось, что кто-то удалил таблицу users в базе данных.
        Резервная копия устарела.

        \vspace{0.5cm}

        \textbf{Что делать?}
        \begin{enumerate}
            \item Разрешить работать с prod базой только \enquote{роботам}
            \item Поменять все пароли и сообщить их только некоторым синьорам
            \item Вести список провинившихся разработчиков
            \item Нанять отдельного специалиста DBA
        \end{enumerate}
    \end{frame}

    \begin{frame}{Кейс 6: Разработчик ушел в отпуск}
        Разработчик завершил выпуск новой функции приложения, но на следующий день ушёл в планируемый отпуск. Остальные сотрудники не знают, как её тестировать,
        настраивать и использовать, из-за отсутствия документации.

        \vspace{0.5cm}

        \textbf{Что делать?}
        \begin{enumerate}
            \item Внести в общие правила пункт \enquote{Не уходить в отпуск без документации}
            \item Нанять отдельного техписателя для написания документации
            \item Ввести в KPI пункт \enquote{покрытие документацией}
            \item Задача выполнена только тогда, когда она в prod и протестирована
        \end{enumerate}
    \end{frame}

    \begin{frame}{Кто занимается DevOps?}
        \begin{center}
            \Large
            Вопрос: если DevOps - это культура для команды, \\
            то зачем существует должность \\
            \enquote{DevOps инженер}?
        \end{center}
    \end{frame}

    \begin{frame}{DevOps vs DevOps инженер}
        \textbf{DevOps — это культура и практики для всей команды}

        \vspace{0.5cm}

        Но на практике кто-то должен:
        \begin{itemize}
            \item Настраивать CI/CD пайплайны
            \item Управлять инфраструктурой
            \item Внедрять практики автоматизации
            \item Быть связующим звеном между Dev и Ops
        \end{itemize}

        \vspace{0.5cm}

        \textbf{DevOps инженер} — специалист, который помогает командам правильно применять DevOps практики
    \end{frame}
