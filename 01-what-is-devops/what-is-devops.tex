        \begin{frame}{Кейс 1. Ошибки при ручном релизе в продакшн}
            \begin{itemize}
                \item Сценарий: Разработчик вручную переносит билд в продакшн в конце недели. Возникают ошибки, отсутствует возможность отката.
                \item Проблема: Отсутствие CI/CD, автоматических проверок, версионирования артефактов.
                \item Решение: CI/CD-пайплайна, автоматизация релизов, хранение стабильных билдов.
            \end{itemize}
        \end{frame}

        \begin{frame}{Кейс 2. Несоответствие тестовой и продакшн среды}
            \begin{itemize}
                \item Сценарий: Тесты происходят на локальной машине или на \textit{каком-то} тестовом сервере.
                \item Проблема: Непоследовательность окружений, отсутствие инфраструктуры как кода.
                \item Решение: Использование IaC, контейнеризация, унификация процессов развертывания.
            \end{itemize}
        \end{frame}
    }

        \setbeamertemplate{frame footer}{

        \begin{frame}{Кейс 3: Недооценка влияния изменений на инфраструктуру}
            \begin{itemize}
                \item Сценарий: Новая функциональность увеличивает нагрузку на дисковое пространство, что приводит к сбоям.
                \item Проблема: Отсутствие взаимодействия между Dev и Ops, отсутствие мониторинга.
                \item Решение: Введение систем мониторинга и алертинга, межфункциональная коммуникация.
            \end{itemize}
        \end{frame}

        \begin{frame}{Кейс 4.  Отсутствие производственного профилирования}
            \begin{itemize}
                \item Сценарий: Новая фича вызывает резкий рост нагрузки, серверы перегружены, метрик нет.
                \item Проблема: Отсутствие метрик, нагрузочного тестирования, оценки производительности.
                \item Решение: Внедрение performance-тестов, логирования, метрик и алертов.
            \end{itemize}
        \end{frame}

        \begin{frame}{Кейс 5: Долгие релизы}
            \begin{itemize}
                \item Сценарий: Разработка фичи завершена быстро, но релиз затягивается из-за ручных процессов.
                \item Проблема: Долгий цикл вывода в продакшн, отсутствие обратной связи, редкие окна для релизов.
                \item Решение: CD систма, feature flags, автоматизация тестирования и проверки безопасности.
            \end{itemize}
        \end{frame}
    }

        \begin{frame}{Кейс 6. Критическая ошибка при работе с продакшн-базой данных}
            \begin{itemize}
                \item Сценарий: Во время миграции разработчик по ошибке удаляет таблицу в продакшн. Нет резервной копии, нет логов.
                \item Проблема: Отсутствие безопасных сред, процесса валидации, культуры постинцидентного анализа, принципа минимальных привилегий.
                \item Решение: Использование изолированных окружений, code review для миграций, внедрение post-mortem практики без поиска виноватых.
            \end{itemize}
        \end{frame}

        \begin{frame}{Кейс 7. Отсутствие эксплуатационной документации}
            \begin{itemize}
                \item Сценарий: После релиза новой функциональности команда не имеет инструкции по использованию и тестированию. Разработчик недоступен.
                \item Проблема: Отсутствие документации, концентрация знаний у отдельных лиц.
                \item Решение: Поддержка актуальной документации, knowledge sharing, встраивание требований к документации в Definition of Done.
            \end{itemize}
        \end{frame}
    }

    \begin{frame}{DevOps - это культура}
        \begin{itemize}
            \item Совместная ответственность Dev и Ops (+QA, Sec)
            \item Автоматизация сборки, тестирования и доставки
            \item Повторяемая инфраструктура как код
            \item Быстрая и надёжная обратная связь
            \item Прозрачность процессов
        \end{itemize}
    \end{frame}

    \begin{frame}{Ключевые практики DevOps}
        \begin{itemize}
            \item CI/CD
            \item Infrastructure as Code (IaC), \href{https://www.cloudbees.com/blog/what-is-everything-as-code-eac\#why-should-you-adopt-everything-as-code}{Everything as Code (EaC)}
            \item Monitoring, Observability
            \item Культура сотрудничества: shared responsibility, blameless postmortems
        \end{itemize}
    \end{frame}

    \begin{frame}{Заключение}
        \begin{itemize}
            \item DevOps — это культура, а не просто инструменты или практики.
            \item Решение системных проблем требует комплексного подхода.
            \item DevOps объединяет разработку, тестирование, эксплуатацию и безопасность.
            \item Внедрение DevOps — это постепенный процесс.
        \end{itemize}
