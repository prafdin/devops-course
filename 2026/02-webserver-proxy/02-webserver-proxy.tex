    \begin{frame}{Что такое веб-сервер?}
        \textbf{HTTP веб-сервер} - программное обеспечение, которое принимает HTTP-запросы от \textbf{клиентов}, обрабатывает и возвращает HTTP-ответы
        
        \vspace{0.5cm}
        
        \textbf{Принцип работы:}
        \begin{enumerate}
            \item Клиент отправляет HTTP-запрос к \textbf{ресурсу}, указывая его \textbf{идентификатор}
            \item Веб-сервер \textbf{обрабатывает} запрос и возвращает HTTP ответ
        \end{enumerate}
        
        \vspace{0.5cm}
        
        \textbf{Примеры веб-серверов:}
        \begin{itemize}
            \item Apache
            \item Nginx
            \item Python http.server
        \end{itemize}
    \end{frame}

    \begin{frame}{Клиенты веб-серверов}
        \textbf{Примеры клиентов веб-серверов:}
        \begin{itemize}
            \item Браузер
            \item Утилиты командной строки (curl, wget)
            \item Другой веб-сервер (reverse proxy)
            \item Серверное приложение (backend приложение)
        \end{itemize}
    \end{frame}

    \begin{frame}{Ресурсы предоставляемые веб-сервером}
        Ресурс - абстрактная сущность, идентифицируемая URL, доступ к которой предоставляется через веб-сервер.

        \textbf{Примеры ресурсов:}
        \begin{itemize}
            \item Файл (HTML файл готовый или обработанный, бинарные файлы)
            \item Динамически сгенерированные данные
            \item Записи в БД
            \item Состояние сервиса
            \item Проксируемые ресурсы другого сервиса
        \end{itemize}
    \end{frame}

    \begin{frame}{Прокси}
        \textbf{Прокси сервер} — сервер-посредник между клиентом (кто запрашивает ресурс) и сервером (кто предоставляет ресурс)
        
        \vspace{0.5cm}
        
        \begin{columns}[t]
            \begin{column}[t]{0.465\textwidth}
                \textbf{Forward proxy:}
                \begin{itemize}
                    \item Клиент → Прокси → Интернет
                    \item Скрывает клиента от сервера
                    \item Пример: корпоративный прокси
                \end{itemize}
            \end{column}
            
            \begin{column}[t]{0.465\textwidth}
                \textbf{Reverse proxy:}
                \begin{itemize}
                    \item Интернет → Прокси → Сервер
                    \item Скрывает сервер от клиента
                    \item Пример: nginx перед приложением
                \end{itemize}
            \end{column}
        \end{columns}
    \end{frame}
